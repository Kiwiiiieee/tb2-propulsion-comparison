%% TB2 ICE Mission Calibrated with Masood and Wei (2012)
% This script uses the published fixed-wing UAV benchmark from:
% Khayyam Masood and Zhang Wei,
% "Detailed Flight Performance Analysis of a Fixed Wing UAV",
% AIAA 2012-2595, DOI: 10.2514/6.2012-2595
%
% Workflow:
% 1) Rebuild the paper's benchmark aircraft and 9-segment mission
% 2) Compare our simplified mission model against the published results
% 3) Calibrate the fuel-consumption model against the paper
% 4) Compare variable-atmosphere and fixed-atmosphere mission results
% 5) Transfer that calibration to the TB2-class specific ISR mission
% 6) Plot stage-normalized results so each phase is equally readable

clear; clc; close all;

dt = 1.0; % s

%% 1) Paper benchmark data
paper = buildMasoodWeiBenchmark();
paper.variableAtmosphere = makeAtmosphereModel(paper.aircraft, 'variable', 0);
paperSimUncal = simulateSegmentedMission(paper.mission, paper.aircraft, dt, paper.variableAtmosphere);
paper.calibration.bsfcScale = paper.published.totalFuelConsumed_kg / paperSimUncal.fuelUsed(end);
paper.aircraft.BSFC = paper.aircraft.BSFC * paper.calibration.bsfcScale;
paperSimVariable = simulateSegmentedMission(paper.mission, paper.aircraft, dt, paper.variableAtmosphere);
paper.fixedAtmosphere = makeAtmosphereModel(paper.aircraft, 'fixed', mean(paperSimVariable.h));
paperSimFixed = simulateSegmentedMission(paper.mission, paper.aircraft, dt, paper.fixedAtmosphere);

paperTables = buildPaperComparisonTables(paper, paperSimVariable, paperSimUncal);
paperStageVariableTable = buildStageMetricMatrix(paperSimVariable.stageSummary);
paperStageFixedTable = buildStageMetricMatrix(paperSimFixed.stageSummary);
paperSensitivityTable = buildAtmosphereSensitivityTable(paperSimVariable, paperSimFixed, paper.fixedAtmosphere.referenceAltitude_m, 'Paper Benchmark');
paperOverallComparison = buildOverallComparisonTable(paper, paperSimVariable, paper.fixedAtmosphere);
paperSensitivityDecision = summarizeAtmosphereImpact(paperSensitivityTable);

%% 2) TB2-class mission with transferred calibration
study = struct();
study.missionName = 'TB2-Class ISR Mission Calibrated by Masood-Wei Benchmark';
study.cruiseAltitude = 3000;      % m
study.outboundRange_km = 120;     % km
study.inboundRange_km = 120;      % km
study.loiterDuration_min = 240;   % min
study.climbDuration_s = 600;
study.descentDuration_s = 700;
study.normalizeStages = true;

tb2 = buildTB2Model();
tb2.BSFC = tb2.BSFC * paper.calibration.bsfcScale;

VstallSL_tb2 = sqrt(2 * tb2.mass0 * tb2.g / (tb2.rhoSL * tb2.wingArea * tb2.CLmax));
VTO_tb2 = 1.20 * VstallSL_tb2;
Vapp_tb2 = 1.30 * VstallSL_tb2;
Vmax_tb2 = tb2.maxSpeedKTAS * 0.514444;

VbestRange_tb2 = computeBestRangeCruiseSpeed(tb2, study.cruiseAltitude, tb2.mass0, Vmax_tb2);
VbestEndurance_tb2 = computeBestEnduranceCruiseSpeed(tb2, study.cruiseAltitude, tb2.mass0, Vmax_tb2);
opt_tb2 = optimizeMissionSpeeds(tb2, study, VstallSL_tb2, Vmax_tb2);
mission_tb2 = buildSpecificMissionDefinition(study, VstallSL_tb2, VTO_tb2, Vapp_tb2, opt_tb2);
tb2.variableAtmosphere = makeAtmosphereModel(tb2, 'variable', 0);
tb2SimVariable = simulateSegmentedMission(mission_tb2, tb2, dt, tb2.variableAtmosphere);
tb2.fixedAtmosphere = makeAtmosphereModel(tb2, 'fixed', mean(tb2SimVariable.h));
tb2SimFixed = simulateSegmentedMission(mission_tb2, tb2, dt, tb2.fixedAtmosphere);

tb2StageVariableTable = buildStageMetricMatrix(tb2SimVariable.stageSummary);
tb2StageFixedTable = buildStageMetricMatrix(tb2SimFixed.stageSummary);
tb2SensitivityTable = buildAtmosphereSensitivityTable(tb2SimVariable, tb2SimFixed, tb2.fixedAtmosphere.referenceAltitude_m, 'TB2 Mission');
tb2SensitivityDecision = summarizeAtmosphereImpact(tb2SensitivityTable);
tb2MissionOverviewTable = buildMissionOverviewTable(tb2, study, opt_tb2, tb2SimVariable, tb2SimFixed, tb2.fixedAtmosphere);

%% Local functions
function paper = buildMasoodWeiBenchmark()
paper = struct();
paper.reference = 'Masood and Wei (2012), AIAA 2012-2595';

% Aircraft data extracted from Tables 1-3 and the text
p = struct();
p.g = 9.81;
p.rhoSL = 1.225;
p.T0 = 288.15;
p.P0 = 101325;
p.R = 287.05;
p.gammaAir = 1.4;
p.mass0 = 540;
p.fuelMass0 = 110;
p.dryMass = p.mass0 - p.fuelMass0;
p.wingspan = 9.0;           % inferred from rectangular wing, area 9 m^2, chord 1 m
p.length = 6.3;
p.wingArea = 9.0;
p.CLmax = 1.55;             % approximated from paper's NACA4412 whole-aircraft curve
p.CD0 = 0.035;              % approximated from paper's CFD drag trend
p.oswald = 0.80;
p.aspectRatio = p.wingspan^2 / p.wingArea;
p.kInduced = 1 / (pi * p.oswald * p.aspectRatio);
p.enginePowerHP = 100;
p.enginePowerMax = p.enginePowerHP * 745.7;
p.etaProp = 0.72;           % starting value before BSFC calibration
p.BSFC = 0.32;              % base value, later calibrated to the paper
p.LHV = 43e6;
p.CO2factor = 3.16;
p.idlePowerFrac = 0.18;
p.CLalpha = 5.7;

paper.aircraft = p;

% Published benchmark mission from Table 5
dur_hr = [0.41 2.04 0.17 0.15 1.00 0.15 0.20 1.88 0.82];
dur_s = dur_hr * 3600;
segRange_km = [54 450 28 30 0 30 30.5 450 111.5];
initH_km = [0.01 7.0 7.0 5.0 5.0 5.0 5.0 8.0 8.0];
finalH_km = [7.0 7.0 5.0 5.0 5.0 5.0 8.0 8.0 0.01];
publishedVel_kmh = [NaN 220 NaN 200 200 200 NaN 240 NaN];

% Where the paper does not tabulate speed directly, derive an average from
% segment range and duration. This is documented so it is report-ready.
derivedVel_kmh = segRange_km ./ max(dur_hr, eps);
derivedVel_kmh(5) = 200; % loiter speed explicitly given in Table 5
V_kmh = publishedVel_kmh;
V_kmh(isnan(V_kmh)) = derivedVel_kmh(isnan(V_kmh));

names = {'Climb','Cruise Out','Descent 1','Cruise Before Loiter','Loiter', ...
         'Cruise After Loiter','Climb 2','Cruise Back','Descent 2'};

for i = 1:numel(names)
    mission(i) = struct( ... %#ok<AGROW>
        'name', names{i}, ...
        'duration', dur_s(i), ...
        'h0', initH_km(i) * 1000, ...
        'h1', finalH_km(i) * 1000, ...
        'V0', V_kmh(i) / 3.6, ...
        'V1', V_kmh(i) / 3.6, ...
        'mode', ternary(strcmp(names{i}, 'Loiter'), 'loiter', 'straight'), ...
        'loiterRadius', ternary(strcmp(names{i}, 'Loiter'), 700, 0));
end

paper.mission = mission;
paper.phaseNames = names;
paper.published = struct();
paper.published.segmentDuration_hr = dur_hr(:);
paper.published.segmentRange_km = segRange_km(:);
paper.published.initialHeight_km = initH_km(:);
paper.published.finalHeight_km = finalH_km(:);
paper.published.cruiseVelocity_kmh = V_kmh(:);
paper.published.stageFuel_kg = [7 24.5 2.5 2 13.5 2 3.5 28.75 12.3].';
paper.published.initialMass_kg = [540 533 503.5 501 499 485.5 483.5 480 451.25].';
paper.published.finalMass_kg = [533 503.5 501 499 485.5 483.5 480 451.25 439].';
paper.published.totalTime_hr = 6.82;
paper.published.totalRange_km = 1184;
paper.published.totalFuelConsumed_kg = 101;
paper.published.remainingFuel_kg = 9;
paper.published.cumulativeStageEnd_hr = cumsum(dur_hr(:));
paper.notes = [ ...
    "Loiter segment speed is explicitly reported as 200 km/hr."; ...
    "The paper's loiter segment range row appears inconsistent with the total range row; here we treat loiter as near-zero net downrange with finite flown path length."; ...
    "Missing segment cruise speeds are reconstructed from published stage range divided by stage duration." ...
    ];
end

function tables = buildPaperComparisonTables(paper, simCal, simUncal)
names = ["Total Time (hr)"; "Total Range (km)"; "Fuel Used (kg)"; "Remaining Fuel (kg)"; "Average Fuel Rate (kg/hr)"];
published = [ ...
    paper.published.totalTime_hr; ...
    paper.published.totalRange_km; ...
    paper.published.totalFuelConsumed_kg; ...
    paper.published.remainingFuel_kg; ...
    paper.published.totalFuelConsumed_kg / paper.published.totalTime_hr];
uncal = [ ...
    simUncal.t(end) / 3600; ...
    simUncal.stageSummary.CumulativeDownrange_km(end); ...
    simUncal.fuelUsed(end); ...
    simUncal.fuelMass(end); ...
    simUncal.fuelUsed(end) / (simUncal.t(end) / 3600)];
cal = [ ...
    simCal.t(end) / 3600; ...
    simCal.stageSummary.CumulativeDownrange_km(end); ...
    simCal.fuelUsed(end); ...
    simCal.fuelMass(end); ...
    simCal.fuelUsed(end) / (simCal.t(end) / 3600)];

total = table(names, published, uncal, cal, ...
    100 * (cal - published) ./ max(abs(published), eps), ...
    'VariableNames', {'Metric','Published','SimulatedUncalibrated','SimulatedCalibrated','PercentError_Calibrated'});

stageNames = string({paper.mission.name}).';
stages = table( ...
    stageNames, ...
    paper.published.segmentDuration_hr, ...
    simCal.stageSummary.Duration_hr, ...
    paper.published.segmentRange_km, ...
    simCal.stageSummary.Downrange_km, ...
    paper.published.stageFuel_kg, ...
    simCal.stageSummary.FuelUsed_kg, ...
    100 * (simCal.stageSummary.FuelUsed_kg - paper.published.stageFuel_kg) ./ max(abs(paper.published.stageFuel_kg), eps), ...
    'VariableNames', {'Stage','PublishedDuration_hr','SimDuration_hr','PublishedRange_km','SimDownrange_km','PublishedFuel_kg','SimFuel_kg','FuelPercentError'});

tables = struct();
tables.total = total;
tables.stages = stages;
end

function VbestRange = computeBestRangeCruiseSpeed(p, cruiseAltitude, massGuess, Vmax)
[~, ~, rhoCruise, ~] = isaTroposphere(cruiseAltitude, p);
VstallCruise = sqrt(2 * massGuess * p.g / (rhoCruise * p.wingArea * p.CLmax));
VminSearch = max(1.35 * VstallCruise, 22);
VmaxSearch = min(0.95 * Vmax, 60);
if VmaxSearch <= VminSearch
    VmaxSearch = VminSearch + 5;
end
Vgrid = linspace(VminSearch, VmaxSearch, 300);
dragMetric = zeros(size(Vgrid));
for i = 1:numel(Vgrid)
    perf = computeSteadyLevelMetrics(p, Vgrid(i), cruiseAltitude, massGuess);
    dragMetric(i) = perf.drag;
end
[~, idxBest] = min(dragMetric);
VbestRange = Vgrid(idxBest);
end

function VbestEndurance = computeBestEnduranceCruiseSpeed(p, cruiseAltitude, massGuess, Vmax)
[~, ~, rhoCruise, ~] = isaTroposphere(cruiseAltitude, p);
VstallCruise = sqrt(2 * massGuess * p.g / (rhoCruise * p.wingArea * p.CLmax));
VminSearch = max(1.25 * VstallCruise, 20);
VmaxSearch = min(0.92 * Vmax, 55);
if VmaxSearch <= VminSearch
    VmaxSearch = VminSearch + 5;
end
Vgrid = linspace(VminSearch, VmaxSearch, 300);
powerMetric = zeros(size(Vgrid));
for i = 1:numel(Vgrid)
    perf = computeSteadyLevelMetrics(p, Vgrid(i), cruiseAltitude, massGuess);
    powerMetric(i) = perf.powerUsed;
end
[~, idxBest] = min(powerMetric);
VbestEndurance = Vgrid(idxBest);
end

function optSummary = optimizeMissionSpeeds(p, study, VstallSL, Vmax)
massGuess = p.mass0 - 0.30 * p.fuelMass0;
[~, ~, rhoCruise, ~] = isaTroposphere(study.cruiseAltitude, p);
VstallCruise = sqrt(2 * massGuess * p.g / (rhoCruise * p.wingArea * p.CLmax));
transitCandidates = linspace(max(1.30 * VstallCruise, 24), min(0.95 * Vmax, 58), 120);
loiterCandidates = linspace(max(1.20 * VstallCruise, 22), min(0.85 * Vmax, 50), 120);
transitFuelPerKm = zeros(size(transitCandidates));
loiterFuelPerHour = zeros(size(loiterCandidates));
for i = 1:numel(transitCandidates)
    perf = computeSteadyLevelMetrics(p, transitCandidates(i), study.cruiseAltitude, massGuess);
    transitFuelPerKm(i) = 1000 * perf.fuelFlow / max(transitCandidates(i), eps);
end
for i = 1:numel(loiterCandidates)
    perf = computeSteadyLevelMetrics(p, loiterCandidates(i), study.cruiseAltitude, massGuess);
    loiterFuelPerHour(i) = 3600 * perf.fuelFlow;
end
[~, idxTransit] = min(transitFuelPerKm);
[~, idxLoiter] = min(loiterFuelPerHour);
optSummary = struct();
optSummary.Vtransit = transitCandidates(idxTransit);
optSummary.Vloiter = loiterCandidates(idxLoiter);
optSummary.transitCandidates = transitCandidates;
optSummary.loiterCandidates = loiterCandidates;
optSummary.transitFuelPerKm = transitFuelPerKm;
optSummary.loiterFuelPerHour = loiterFuelPerHour;
optSummary.VstallCruise = VstallCruise;
optSummary.VstallSL = VstallSL;
end

function perf = computeSteadyLevelMetrics(p, V, altitude, massGuess)
[~, ~, rho, ~] = isaTroposphere(altitude, p);
q = 0.5 * rho * V^2;
lift = massGuess * p.g;
CL = lift / max(q * p.wingArea, eps);
CD = p.CD0 + p.kInduced * CL^2;
drag = q * p.wingArea * CD;
thrustReq = drag;
powerReq = thrustReq * V / p.etaProp;
powerUsed = min(max(powerReq, p.idlePowerFrac * p.enginePowerMax), p.enginePowerMax);
fuelFlow = p.BSFC * (powerUsed / 1000) / 3600;
perf = struct('CL',CL,'CD',CD,'drag',drag,'thrustReq',thrustReq,'powerReq',powerReq,'powerUsed',powerUsed,'fuelFlow',fuelFlow);
end

function sim = simulateSegmentedMission(mission, p, dt, atmModel)
[t, V, h, phaseIdx, phaseNames, phaseEdges, modeIdx, loiterRadius] = buildMissionProfile(mission, dt);

V = max(V, 0.1);
h = max(h, 0);
hdot = gradient(h, dt);
gammaPath = asin(max(min(hdot ./ max(V, 0.1), 0.95), -0.95));
Vh = sqrt(max(V.^2 - hdot.^2, 0));
aLong = gradient(V, dt);

[tempK, pressurePa, rho, aSound] = evaluateAtmosphere(h, p, atmModel);
mach = V ./ aSound;

N = numel(t);
x = zeros(N,1);
y = zeros(N,1);
pathS = zeros(N,1);
heading = zeros(N,1);

for k = 1:(N-1)
    pathS(k+1) = pathS(k) + Vh(k) * dt;
    if modeIdx(k) == 2
        omega = Vh(k) / max(loiterRadius(k), 1);
        heading(k+1) = heading(k) + omega * dt;
    else
        heading(k+1) = 0;
    end
    x(k+1) = x(k) + Vh(k) * cos(heading(k)) * dt;
    y(k+1) = y(k) + Vh(k) * sin(heading(k)) * dt;
end

mass = zeros(N,1);
fuelMass = zeros(N,1);
fuelUsed = zeros(N,1);
CL = zeros(N,1);
CD = zeros(N,1);
Lift = zeros(N,1);
Drag = zeros(N,1);
ThrustReq = zeros(N,1);
PowerReq = zeros(N,1);
PowerUsed = zeros(N,1);
fuelFlow = zeros(N,1);
fuelEnergyRate = zeros(N,1);
fuelEnergyCum = zeros(N,1);
CO2cum = zeros(N,1);
LD = zeros(N,1);
thrustAvailable = zeros(N,1);

mass(1) = p.mass0;
fuelMass(1) = p.fuelMass0;

for k = 1:N
    mNow = mass(k);
    q = 0.5 * rho(k) * V(k)^2;
    Lift(k) = mNow * p.g * cos(gammaPath(k));
    CL(k) = Lift(k) / max(q * p.wingArea, eps);
    CD(k) = p.CD0 + p.kInduced * CL(k)^2;
    Drag(k) = q * p.wingArea * CD(k);

    thrustRaw = Drag(k) + mNow * aLong(k) + mNow * p.g * sin(gammaPath(k));
    ThrustReq(k) = max(thrustRaw, 0);
    PowerReq(k) = ThrustReq(k) * V(k) / p.etaProp;
    thrustAvailable(k) = p.etaProp * p.enginePowerMax / max(V(k), 8);
    PowerUsed(k) = min(max(PowerReq(k), p.idlePowerFrac * p.enginePowerMax), p.enginePowerMax);
    fuelFlow(k) = p.BSFC * (PowerUsed(k) / 1000) / 3600;
    fuelEnergyRate(k) = fuelFlow(k) * p.LHV;
    LD(k) = Lift(k) / max(Drag(k), 1e-6);

    if k < N
        fuelUsed(k+1) = fuelUsed(k) + fuelFlow(k) * dt;
        fuelMass(k+1) = max(p.fuelMass0 - fuelUsed(k+1), 0);
        mass(k+1) = p.dryMass + fuelMass(k+1);
        fuelEnergyCum(k+1) = fuelEnergyCum(k) + fuelEnergyRate(k) * dt;
        CO2cum(k+1) = fuelUsed(k+1) * p.CO2factor;
    end
end

fuelEnergyCumMJ = fuelEnergyCum / 1e6;
VstallTime = sqrt(2 * mass * p.g ./ max(rho * p.wingArea * p.CLmax, 1e-6));

numPhases = numel(phaseNames);
stageDuration_hr = zeros(numPhases,1);
stagePathKm = zeros(numPhases,1);
stageDownrangeKm = zeros(numPhases,1);
stageFuelKg = zeros(numPhases,1);
stageEnergyMJ = zeros(numPhases,1);
stageMeanPowerkW = zeros(numPhases,1);
stageMeanThrustN = zeros(numPhases,1);
stageMeanAccel = zeros(numPhases,1);
stageMeanCL = zeros(numPhases,1);
stageMeanCD = zeros(numPhases,1);
stageMeanFuelFlow = zeros(numPhases,1);

for i = 1:numPhases
    idx = find(phaseIdx == i);
    tStart = phaseEdges(i);
    tEnd = phaseEdges(i + 1);
    xStart = interp1(t, x, tStart, 'linear', 'extrap');
    xEnd = interp1(t, x, tEnd, 'linear', 'extrap');
    sStart = interp1(t, pathS, tStart, 'linear', 'extrap');
    sEnd = interp1(t, pathS, tEnd, 'linear', 'extrap');
    fuelStart = interp1(t, fuelUsed, tStart, 'linear', 'extrap');
    fuelEnd = interp1(t, fuelUsed, tEnd, 'linear', 'extrap');
    energyStart = interp1(t, fuelEnergyCumMJ, tStart, 'linear', 'extrap');
    energyEnd = interp1(t, fuelEnergyCumMJ, tEnd, 'linear', 'extrap');

    stageDuration_hr(i) = (tEnd - tStart) / 3600;
    stagePathKm(i) = (sEnd - sStart) / 1000;
    stageDownrangeKm(i) = (xEnd - xStart) / 1000;
    stageFuelKg(i) = fuelEnd - fuelStart;
    stageEnergyMJ(i) = energyEnd - energyStart;
    stageMeanPowerkW(i) = mean(PowerUsed(idx)) / 1000;
    stageMeanThrustN(i) = mean(ThrustReq(idx));
    stageMeanAccel(i) = mean(aLong(idx));
    stageMeanCL(i) = mean(CL(idx));
    stageMeanCD(i) = mean(CD(idx));
    stageMeanFuelFlow(i) = mean(fuelFlow(idx));
end
stageSummary = table( ...
    string(phaseNames(:)), ...
    stageDuration_hr, ...
    cumsum(stageDuration_hr), ...
    stagePathKm, ...
    cumsum(stagePathKm), ...
    stageDownrangeKm, ...
    cumsum(stageDownrangeKm), ...
    stageFuelKg, ...
    cumsum(stageFuelKg), ...
    stageEnergyMJ, ...
    stageMeanPowerkW, ...
    stageMeanThrustN, ...
    stageMeanAccel, ...
    stageMeanCL, ...
    stageMeanCD, ...
    stageMeanFuelFlow, ...
    'VariableNames', {'Stage','Duration_hr','CumulativeTime_hr','PathLength_km','CumulativePath_km','Downrange_km','CumulativeDownrange_km','FuelUsed_kg','CumulativeFuelUsed_kg','FuelEnergy_MJ','MeanPower_kW','MeanThrust_N','MeanAccel_mps2','MeanCL','MeanCD','MeanFuelFlow_kgs'});

sim = struct();
sim.t = t;
sim.V = V;
sim.h = h;
sim.phaseIdx = phaseIdx;
sim.phaseNames = phaseNames;
sim.phaseEdges = phaseEdges;
sim.x = x;
sim.y = y;
sim.pathS = pathS;
sim.gammaPath = gammaPath;
sim.aLong = aLong;
sim.tempK = tempK;
sim.pressurePa = pressurePa;
sim.rho = rho;
sim.aSound = aSound;
sim.mach = mach;
sim.mass = mass;
sim.fuelMass = fuelMass;
sim.fuelUsed = fuelUsed;
sim.CL = CL;
sim.CD = CD;
sim.Lift = Lift;
sim.Drag = Drag;
sim.ThrustReq = ThrustReq;
sim.PowerReq = PowerReq;
sim.PowerUsed = PowerUsed;
sim.fuelFlow = fuelFlow;
sim.fuelEnergyRate = fuelEnergyRate;
sim.fuelEnergyCum = fuelEnergyCum;
sim.fuelEnergyCumMJ = fuelEnergyCumMJ;
sim.CO2cum = CO2cum;
sim.LD = LD;
sim.thrustAvailable = thrustAvailable;
sim.VstallTime = VstallTime;
sim.stageSummary = stageSummary;
end

function atmModel = makeAtmosphereModel(p, mode, referenceAltitude_m)
mode = lower(mode);
atmModel = struct();
atmModel.mode = mode;
switch mode
    case 'variable'
        atmModel.referenceAltitude_m = NaN;
        atmModel.temperatureK = NaN;
        atmModel.pressurePa = NaN;
        atmModel.density = NaN;
        atmModel.speedOfSound = NaN;
    case 'fixed'
        [Tref, Pref, rhoref, aref] = isaTroposphere(referenceAltitude_m, p);
        atmModel.referenceAltitude_m = referenceAltitude_m;
        atmModel.temperatureK = Tref;
        atmModel.pressurePa = Pref;
        atmModel.density = rhoref;
        atmModel.speedOfSound = aref;
    otherwise
        error('Unknown atmosphere mode "%s". Use "variable" or "fixed".', mode);
end
end

function [T, P, rho, aSound] = evaluateAtmosphere(h, p, atmModel)
if nargin < 3 || isempty(atmModel) || ~isfield(atmModel, 'mode') || strcmpi(atmModel.mode, 'variable')
    [T, P, rho, aSound] = isaTroposphere(h, p);
    return;
end

switch lower(atmModel.mode)
    case 'fixed'
        T = atmModel.temperatureK * ones(size(h));
        P = atmModel.pressurePa * ones(size(h));
        rho = atmModel.density * ones(size(h));
        aSound = atmModel.speedOfSound * ones(size(h));
    otherwise
        [T, P, rho, aSound] = isaTroposphere(h, p);
end
end

function decision = summarizeAtmosphereImpact(sensitivityTable)
keyMetrics = ["MissionTime_hr","PathLength_km","FuelUsed_kg","Downrange_km"];

if maxKeyPercentDifference < 2
    recommendation = "Fixed atmosphere is essentially identical for the key outputs.";
elseif maxKeyPercentDifference < 5
    recommendation = "Fixed atmosphere is acceptable for early-stage analysis, with only small deviations.";
else
    recommendation = "Altitude-varying atmosphere should be retained because the impact is no longer negligible.";
end

decision = struct();
decision.keyMetrics = keyMetrics;
decision.maxKeyPercentDifference = maxKeyPercentDifference;
decision.recommendation = recommendation;
end

function normalized = buildNormalizedStageProfiles(sim)
numStages = numel(sim.phaseNames);
equalStageTime = zeros(size(sim.t));
localStageTime = zeros(size(sim.t));
for i = 1:numStages
    idx = find(sim.phaseIdx == i);
    tStart = sim.phaseEdges(i);
    tEnd = sim.phaseEdges(i + 1);
    denom = max(tEnd - tStart, eps);
    localStageTime(idx) = (sim.t(idx) - tStart) / denom;
    equalStageTime(idx) = (i - 1) + localStageTime(idx);
end
normalized = struct();
normalized.numStages = numStages;
normalized.equalStageTime = equalStageTime;
normalized.localStageTime = localStageTime;
normalized.V = sim.V;
normalized.fuelFlow = sim.fuelFlow;
normalized.ThrustReq = sim.ThrustReq;
normalized.fuelUsed = sim.fuelUsed;
end

function [t, V, h, phaseIdx, phaseNames, phaseEdges, modeIdx, loiterRadius] = buildMissionProfile(mission, dt)
numPhases = numel(mission);
phaseNames = cell(numPhases,1);
phaseEdges = zeros(numPhases + 1, 1);
t = [];
V = [];
h = [];
phaseIdx = [];
modeIdx = [];
loiterRadius = [];
timeCursor = 0;
phaseEdges(1) = 0;

for i = 1:numPhases
    phaseNames{i} = mission(i).name;
    nPts = round(mission(i).duration / dt) + 1;
    tau = linspace(0, mission(i).duration, nPts).';
    tSeg = timeCursor + tau;
    VSeg = linspace(mission(i).V0, mission(i).V1, nPts).';
    hSeg = linspace(mission(i).h0, mission(i).h1, nPts).';
    modeSeg = ones(nPts,1);
    if isfield(mission(i), 'mode') && strcmpi(mission(i).mode, 'loiter')
        modeSeg(:) = 2;
    end
    radSeg = zeros(nPts,1);
    if isfield(mission(i), 'loiterRadius')
        radSeg(:) = mission(i).loiterRadius;
    end

    if i > 1
        tSeg = tSeg(2:end);
        VSeg = VSeg(2:end);
        hSeg = hSeg(2:end);
        modeSeg = modeSeg(2:end);
        radSeg = radSeg(2:end);
    end

    t = [t; tSeg]; %#ok<AGROW>
    V = [V; VSeg]; %#ok<AGROW>
    h = [h; hSeg]; %#ok<AGROW>
    phaseIdx = [phaseIdx; i * ones(numel(tSeg),1)]; %#ok<AGROW>
    modeIdx = [modeIdx; modeSeg]; %#ok<AGROW>
    loiterRadius = [loiterRadius; radSeg]; %#ok<AGROW>

    timeCursor = timeCursor + mission(i).duration;
    phaseEdges(i + 1) = timeCursor;
end
end

function [T, P, rho, aSound] = isaTroposphere(h, p)
T = zeros(size(h));
P = zeros(size(h));
rho = zeros(size(h));
aSound = zeros(size(h));
for i = 1:numel(h)
    hi = h(i);
    if hi <= 11000
        L = 0.0065;
        T(i) = p.T0 - L * hi;
        P(i) = p.P0 * (T(i) / p.T0)^(p.g / (p.R * L));
    else
        T11 = 216.65;
        P11 = p.P0 * (T11 / p.T0)^(p.g / (p.R * 0.0065));
        T(i) = T11;
        P(i) = P11 * exp(-p.g * (hi - 11000) / (p.R * T11));
    end
    rho(i) = P(i) / (p.R * T(i));
    aSound(i) = sqrt(p.gammaAir * p.R * T(i));
end
end

end
end
